```dart
  return {'status': false, 'msg': e.toString()};

// HTTP层中
try {
  var res = await Request().get(Api.searchTrending, data: {'limit': limit});
  if (res.data['code'] == 0) {
    var model = SearchTrendingModel.fromJson(res.data['data']);
    return {
      'status': true,
      'data': model,
    };
  } else {
    return {
      'status': false,
      'data': null,
      'msg': res.data['message'] ?? '请求错误',
    };
  }
} catch (e) {
  return {
    'status': false,
    'data': null,
    'msg': e.toString(),
  };
}
```
